cmake_minimum_required(VERSION 3.28...4.5)
project(a08_junction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

set(real "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include")
set(junc "${CMAKE_BINARY_DIR}-junction")
if(NOT EXISTS "${junc}/jt/jt.hpp")
  if(WIN32)
    file(TO_NATIVE_PATH "${junc}" nj)
    file(TO_NATIVE_PATH "${real}" nr)
    execute_process(COMMAND cmd /c mklink /J "${nj}" "${nr}" RESULT_VARIABLE rc OUTPUT_VARIABLE o ERROR_VARIABLE o)
    message(STATUS "mklink /J: rc=${rc} ${o}")
  else()
    file(CREATE_LINK "${real}" "${junc}" RESULT rc SYMBOLIC)   # port: Linux analogue (directory symlink)
    message(STATUS "symlink ${junc} -> ${real}: ${rc}")
  endif()
endif()
add_library(jt_dotdot INTERFACE IMPORTED)
set_target_properties(jt_dotdot PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sub/../include")
add_library(jt_junc INTERFACE IMPORTED)
set_target_properties(jt_junc PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${junc}")
add_header_units(hu_jt  HEADERS jt/jt.hpp LINK_LIBRARIES jt_dotdot)   # HU via '..' path
option(A08_SINGLE "port: only the set hu_jt (no second set declaring the same file)" OFF)
if(NOT A08_SINGLE)
  add_header_units(hu_jt2 HEADERS jt/jt.hpp LINK_LIBRARIES jt_junc)   # HU via junction
endif()
add_executable(a08_junc main.cpp)          # consumer resolves through the junction, HU via '..'
target_link_libraries(a08_junc PRIVATE jt_junc)
target_import_header_units(a08_junc hu_jt)
add_executable(a08_dotdot main2.cpp)       # consumer resolves through '..', HU via '..'
target_link_libraries(a08_dotdot PRIVATE jt_dotdot)
target_import_header_units(a08_dotdot hu_jt)
if(NOT A08_SINGLE)
  add_executable(a08_junc2 main3.cpp)      # consumer resolves through '..', HU via junction
  target_link_libraries(a08_junc2 PRIVATE jt_dotdot)
  target_import_header_units(a08_junc2 hu_jt2)
endif()
