cmake_minimum_required(VERSION 3.28...4.5)
project(a05_compat LANGUAGES CXX)
set(A05_MODE "features" CACHE STRING "features|extoff")
if(A05_MODE STREQUAL "extoff")
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_EXTENSIONS OFF)
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)
if(A05_MODE STREQUAL "features")
  # no CMAKE_CXX_STANDARD: the modern target_compile_features() style
  add_header_units(hu_v HEADERS vector string CXX_STANDARD 20)
else()
  add_header_units(hu_v HEADERS vector string)
endif()
add_executable(a05 main.cpp)
if(A05_MODE STREQUAL "features")
  target_compile_features(a05 PRIVATE cxx_std_20)
endif()
target_import_header_units(a05 hu_v)
