cmake_minimum_required(VERSION 3.28...4.5)
project(a06_longpath LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

set(A06_TOTAL 255 CACHE STRING "length of the header unit's object path")
string(LENGTH "${CMAKE_BINARY_DIR}/CMakeFiles/hu_long.dir/hu/longlib/.hpp.obj" base)
math(EXPR k "${A06_TOTAL} - ${base}")
string(REPEAT "x" ${k} nm)
set(gen "${CMAKE_BINARY_DIR}-vendor/include")
file(CONFIGURE OUTPUT "${gen}/longlib/${nm}.hpp" CONTENT "#pragma once\ninline int longfn() { return 5; }\n")
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/gen/main.cpp" CONTENT "import <longlib/${nm}.hpp>;\nint main() { return longfn() == 5 ? 0 : 1; }\n")
add_library(longlib INTERFACE IMPORTED)
set_target_properties(longlib PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${gen}")
add_header_units(hu_long HEADERS longlib/${nm}.hpp LINK_LIBRARIES longlib)
message(STATUS "A06: expected object path (${A06_TOTAL} chars): ${CMAKE_BINARY_DIR}/CMakeFiles/hu_long.dir/hu/longlib/${nm}.hpp.obj")
add_executable(a06 "${CMAKE_BINARY_DIR}/gen/main.cpp")
target_link_libraries(a06 PRIVATE longlib)
target_import_header_units(a06 hu_long)
