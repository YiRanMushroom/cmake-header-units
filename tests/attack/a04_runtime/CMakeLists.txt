cmake_minimum_required(VERSION 3.28...4.5)
project(a04_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

add_header_units(hu_vec HEADERS vector)
# A04_ONLY: the control a04_ok plus one mismatched target, as the configure-time ABI check rejects a mismatch for the
# whole project (one variant per target in run_attack.cmake).
set(A04_ONLY "" CACHE STRING "a04_rt, a04_idl or a04_gdbg: build only a04_ok and this one; none: only a04_ok")
set(a04_targets a04_ok a04_rt a04_idl a04_gdbg)
if(A04_ONLY)
  set(a04_targets a04_ok)
  if(NOT A04_ONLY STREQUAL "none")
    list(APPEND a04_targets ${A04_ONLY})
  endif()
endif()
foreach(t IN LISTS a04_targets)
  add_executable(${t} imp.cpp txt.cpp main.cpp)
  target_import_header_units(${t} hu_vec SOURCES imp.cpp)
endforeach()
if(TARGET a04_rt)
  set_target_properties(a04_rt PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)   # static release CRT in a Debug build
endif()
if(TARGET a04_idl)
  target_compile_definitions(a04_idl PRIVATE _ITERATOR_DEBUG_LEVEL=0)
endif()
if(TARGET a04_gdbg)
  target_compile_definitions(a04_gdbg PRIVATE _GLIBCXX_DEBUG)            # libstdc++ analogue (port; no effect on MSVC STL)
endif()
