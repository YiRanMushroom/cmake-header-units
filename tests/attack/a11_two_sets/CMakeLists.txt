cmake_minimum_required(VERSION 3.28...4.5)
project(a11_two_sets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

include(${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/common/ExternalLibs.cmake)
option(A11_NO_DUP "port: omit the part with <vector> in two sets (GCC allows one header unit per header)" OFF)
if(NOT A11_NO_DUP)
  add_header_units(hu_a HEADERS vector)
  add_header_units(hu_b HEADERS vector string)
  add_executable(a11 a.cpp b.cpp)
  target_import_header_units(a11 hu_a SOURCES a.cpp)
  target_import_header_units(a11 hu_b SOURCES b.cpp)
endif()
add_header_units(hu_fmt HEADERS fmt/format.h LINK_LIBRARIES fmt::fmt)
add_header_units(hu_s HEADERS string)
add_executable(a11_fmt c.cpp d.cpp)
target_link_libraries(a11_fmt PRIVATE fmt::fmt)
target_import_header_units(a11_fmt hu_fmt)
target_import_header_units(a11_fmt hu_s)
enable_testing()
if(NOT A11_NO_DUP)
  add_test(NAME a11 COMMAND a11)
endif()
add_test(NAME a11_fmt COMMAND a11_fmt)
