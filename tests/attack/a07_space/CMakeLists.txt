cmake_minimum_required(VERSION 3.28...4.5)
project(a07_space LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

add_library(sp::sp INTERFACE IMPORTED)
set_target_properties(sp::sp PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor dir/include")
add_header_units(hu_sp HEADERS vector sp/sp.hpp LINK_LIBRARIES sp::sp)
add_executable(a07 main.cpp)
target_link_libraries(a07 PRIVATE sp::sp)
target_import_header_units(a07 hu_sp)
