cmake_minimum_required(VERSION 3.28...4.5)
project(a10_order LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

include(${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/common/ExternalLibs.cmake)
add_header_units(hu_fmt HEADERS fmt/format.h LINK_LIBRARIES fmt::fmt)
add_header_units(hu_vs  HEADERS vector string)
option(A10_ONE_SET "port: give the hu_v probes the set hu_vs instead of a second set declaring <vector>" OFF)
if(A10_ONE_SET)
  set(hu_v_set hu_vs)
else()
  add_header_units(hu_v   HEADERS vector)
  set(hu_v_set hu_v)
endif()
foreach(p incvec_impfmt impfmt_incvec)
  add_library(p_${p} OBJECT EXCLUDE_FROM_ALL ${p}.cpp)
  target_link_libraries(p_${p} PRIVATE fmt::fmt)
  target_import_header_units(p_${p} hu_fmt)
endforeach()
foreach(p impvec_incstring_sameset impvec_impstring)
  add_library(p_${p} OBJECT EXCLUDE_FROM_ALL ${p}.cpp)
  target_import_header_units(p_${p} hu_vs)
endforeach()
foreach(p winh_impvec impvec_winh cstdint_impvec cassert_impvec impvec_cassert)
  add_library(p_${p} OBJECT EXCLUDE_FROM_ALL ${p}.cpp)
  target_import_header_units(p_${p} ${hu_v_set})
endforeach()
add_library(p_gmf OBJECT EXCLUDE_FROM_ALL)
target_sources(p_gmf PUBLIC FILE_SET CXX_MODULES FILES gmf.cppm)
target_import_header_units(p_gmf hu_vs)
