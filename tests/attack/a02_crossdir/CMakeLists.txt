cmake_minimum_required(VERSION 3.28...4.5)
project(a02_crossdir LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

# editable vendored copy OUTSIDE the build tree
set(A02_VENDOR "${CMAKE_BINARY_DIR}-vendor" CACHE PATH "")
if(NOT EXISTS "${A02_VENDOR}/include/tiny2/tiny.hpp")
  file(COPY vendor/include DESTINATION "${A02_VENDOR}")
endif()
add_library(tiny2::tiny INTERFACE IMPORTED GLOBAL)
set_target_properties(tiny2::tiny PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${A02_VENDOR}/include")
add_header_units(hu_tiny HEADERS tiny2/tiny.hpp LINK_LIBRARIES tiny2::tiny)
add_subdirectory(modlib)
add_subdirectory(app)
