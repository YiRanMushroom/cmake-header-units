cmake_minimum_required(VERSION 3.28...4.5)
project(a12_unity_pch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
include(HeaderUnits)

add_header_units(hu_vec HEADERS vector)
add_executable(a12_unity u1.cpp u2.cpp u3.cpp)
set_target_properties(a12_unity PROPERTIES UNITY_BUILD ON)
target_import_header_units(a12_unity hu_vec SOURCES u1.cpp u2.cpp)
add_executable(a12_pch p1.cpp)
target_precompile_headers(a12_pch PRIVATE <string>)
target_import_header_units(a12_pch hu_vec)
