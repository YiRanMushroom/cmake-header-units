cmake_minimum_required(VERSION 3.28...4.5)
project(t04_shared_set LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake")
include(HeaderUnits)

add_header_units(hu_std HEADERS vector string)

add_executable(t04_a a.cpp)
target_import_header_units(t04_a hu_std)
add_executable(t04_b b.cpp)
target_import_header_units(t04_b hu_std)

set(T04_BAD "" CACHE STRING "std|ext|exc|dup: add a consumer that must be rejected at configure time")
if(T04_BAD STREQUAL "std")
  add_executable(t04_bad bad.cpp)
  set_target_properties(t04_bad PROPERTIES CXX_STANDARD 23)
  target_import_header_units(t04_bad hu_std)
elseif(T04_BAD STREQUAL "ext")
  add_executable(t04_bad bad.cpp)
  set_target_properties(t04_bad PROPERTIES CXX_EXTENSIONS OFF)
  target_import_header_units(t04_bad hu_std)
elseif(T04_BAD STREQUAL "exc")
  add_executable(t04_bad bad.cpp)
  target_compile_options(t04_bad PRIVATE -fno-exceptions)
  target_import_header_units(t04_bad hu_std)
elseif(T04_BAD STREQUAL "dup")
  # A second set with the same header, and a TU that would receive both.
  add_header_units(hu_dup HEADERS vector)
  add_executable(t04_bad bad.cpp)
  target_import_header_units(t04_bad hu_std)
  target_import_header_units(t04_bad hu_dup)
endif()

enable_testing()
add_test(NAME t04_a COMMAND t04_a)
add_test(NAME t04_b COMMAND t04_b)
